% Keep adjacent bullet baselines about 1.5 normal lines apart. The same value
% is reused by production rendering and hidden height measurement.
\newlength{\cveventitemspacing}
\setlength{\cveventitemspacing}{0.5\baselineskip}

% Each bullet is an intrinsic vertical unit. This preserves the visual list
% while making every selected item's observed height exactly additive.
\newcommand{\cveventitem}[1]{%
  \par
  \vskip\cveventitemspacing%
  \nointerlineskip
  \vbox{%
    \hsize=\linewidth
    \leftskip=1.5em
    \parindent=0pt
    \noindent
    \llap{\textbullet\hspace{0.6em}}#1\par
  }%
}

\newcommand{\cveventitems}[1]{%
  \def\temp{#1}%
  \ifx\temp\empty
    % do nothing, no items
  \else
    % Every item carries the inter-item spacing before it. Cancel that spacing
    % once so the first item starts immediately while later items stay apart.
    \vspace{-\cveventitemspacing}%
    #1%
    \vspace{\cveventitemspacing}%
  \fi
}
\let\maybeitemize\cveventitems

\newcommand{\maybeindent}[1]{%
  \def\temp{#1}%
  \ifx\temp\empty
  \else
    \par\hspace{1.5em}#1\par
  \fi
}

% creates a stretched box as cv entry headline followed by some paragraphs about
% the work you did

\vspace{-8pt}

\vspace{6pt}

\maybeindent{#5}

\cveventitems{#4}

\vspace{6pt}
}%
}

% Explicit page layouts split event-based sections into atomic render units.
% #1 is the section label, #2 is an optional event diagnostic, #3 and #4 are
% the current-page and fresh-page prefixes, and #5 is section-ending content.
\newif\ifcvexplicitforcefresh
\newcount\cvexplicitlastunitpage
\cvexplicitforcefreshfalse
\newcommand{\cvexplicitnextunitfresh}{%
    \global\cvexplicitforcefreshtrue
}

\NewEnviron{cvexplicitunit}[5]{%
    \par
    \cvsetflowcontentbox{\flowcurrentbox}{#3\BODY#5}%
    \cvsetflowcontentbox{\flowfreshbox}{#4\BODY#5}%

    \dimen0=\dimexpr\ht\flowfreshbox+\dp\flowfreshbox\relax
    \ifdim\dimen0>\textheight
        \if\relax\detokenize{#2}\relax
            \PackageError{FindJobHelper}{FJH_SECTION_PAGE_OVERFLOW: #1}{The complete static section must fit on a fresh page.}%
        \else
            \PackageError{FindJobHelper}{FJH_EVENT_PAGE_OVERFLOW: #1 / #2}{A complete CV event must fit on a fresh page.}%
        \fi
    \fi

    \ifcvexplicitforcefresh
        \global\cvexplicitforcefreshfalse
        \global\cvexplicitlastunitpage=\value{page}
        \nointerlineskip\box\flowfreshbox
    \else
        \ifdim\pagetotal=0pt
            \global\cvexplicitlastunitpage=\value{page}
            \nointerlineskip\box\flowfreshbox
        \else
            \dimen0=\dimexpr\ht\flowcurrentbox+\dp\flowcurrentbox\relax
            \dimen2=\dimexpr\pagegoal-\pagetotal\relax
            \ifdim\dimen0>\dimen2
                \newpage
                \global\cvexplicitlastunitpage=\value{page}
                \nointerlineskip\box\flowfreshbox
            \else
                \global\cvexplicitlastunitpage=\value{page}
                \nointerlineskip\box\flowcurrentbox
            \fi
        \fi
    \fi
}

% #1 is a stable section label used in overflow diagnostics.
\NewEnviron{flowblock}[1]{%
    \par

    \cvsetflowcontentbox{\flowcurrentbox}{\cvflowblockcurrentcontent{\BODY}}%
    \cvsetflowcontentbox{\flowfreshbox}{\cvflowblockfreshcontent{\BODY}}%

    \setlength{\flowheight}{%
        \dimexpr\ht\flowcurrentbox+\dp\flowcurrentbox\relax
    }

    % Remaining vertical space on current page
    \dimen0=\dimexpr\pagegoal-\pagetotal\relax

    % If the complete fit representation fits here, use the measured box.
    \ifdim\flowheight>\dimen0
        % The fresh representation includes page-top spacing, section spacing,
        % and trailing glue. It must fit as one box or generation fails.
        \dimen2=\dimexpr\ht\flowfreshbox+\dp\flowfreshbox\relax
        \ifdim\dimen2>\textheight
            \PackageError{FindJobHelper}{FJH_SECTION_PAGE_OVERFLOW: #1}{The complete section, including fresh-page spacing, must fit on one page.}%
        \fi
        \newpage
        % Insert the measured box directly in vertical mode, preventing TeX
        % from adding an unmeasured interline skip around it.
        \nointerlineskip\box\flowfreshbox
    \else
        % The current-page representation follows the same exact-box rule.
        \nointerlineskip\box\flowcurrentbox
    \fi
}
