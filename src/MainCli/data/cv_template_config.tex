% Each bullet is an intrinsic vertical unit. This preserves the visual list
% while making every selected item's observed height exactly additive.
\newcommand{\cveventitem}[1]{%
  \par
  \vskip3pt%
  \nointerlineskip
  \vbox{%
    \hsize=\linewidth
    \leftskip=1.5em
    \parindent=0pt
    \noindent
    \llap{\textbullet\hspace{0.6em}}#1\par
  }%
}

\newcommand{\cveventitems}[1]{%
  \def\temp{#1}%
  \ifx\temp\empty
    % do nothing, no items
  \else
    #1%
    \vspace{3pt}%
  \fi
}
\let\maybeitemize\cveventitems

\newcommand{\maybeindent}[1]{%
  \def\temp{#1}%
  \ifx\temp\empty
  \else
    \par\hspace{1.5em}#1\par
  \fi
}

% creates a stretched box as cv entry headline followed by some paragraphs about
% the work you did
% param 1:	event time i.e. 2014 or 2011-2014 etc.
% param 2:	event name (what did you do?)
% param 3:	institution (where did you work / study)
% param 4:	list of paragraphs
% param 5:	just text before the bullet points
%
\newcommand{\cvevent}[5]
{
\par
\nointerlineskip
\vbox{%
\hsize=\linewidth
\begin{tabular*}{1\textwidth}{p{13.6cm}  x{3.9cm}}
    \textbf{#2}%
    \if\relax\detokenize{#3}\relax
    \else
        \ - \textcolor{bgcol}{#3}%
    \fi
    &
    \vspace{2.5pt}\textcolor{sectcol}{#1}
\end{tabular*}

\vspace{-8pt}
\textcolor{softcol}{\hrule}
\vspace{6pt}

\maybeindent{#5}

\cveventitems{#4}

\vspace{6pt}
}%
}

% Build an intrinsic production-width box. This primitive intentionally has no
% page-state inspection and never inserts the resulting box into the page stream.
\newcommand{\cvsetflowcontentbox}[2]{%
    \sbox{#1}{%
        \begin{varwidth}{\linewidth}
            #2
        \end{varwidth}%
    }%
}

\newcommand{\cvflowblockfitskip}{%
    \addvspace{\cvflowblockfitskipamount}%
}

\newcommand{\cvflowblocknewpageskip}{%
    \vspace*{1.5\baselineskip}%
}

% The complete representation used when a section fits on the current page.
% Production and measurement must use this exact wrapper so all leading and
% trailing glue is charged once.
\newcommand{\cvflowblockfitcontent}[1]{%
    \cvflowblockfitskip
    #1
    \par
    \hspace{1em plus 1fill}\allowbreak
    \par
}

% Hidden measurement entry points. The caller reads \ht/\dp from
% \cvmeasurementbox and must never call \usebox.
\newcommand{\cvsetmeasurementbox}[1]{%
    \cvsetflowcontentbox{\cvmeasurementbox}{#1}%
}

\newcommand{\cvsetmeasurementsectionbox}[1]{%
    \cvsetflowcontentbox{\cvmeasurementbox}{\cvflowblockfitcontent{#1}}%
    % Production contributes 1pt at each flowblock boundary. Charge it to
    % the section; document-header measurement cancels it for the first one.
    \dimen0=\dimexpr\ht\cvmeasurementbox+\dp\cvmeasurementbox+1pt\relax
    \setbox\cvmeasurementbox=\vbox to \dimen0{\vfil}%
}

\newcommand{\cvmeasurementsentinelevent}{%
    \cvevent{}{}{Personal}{}{}%
}

\newcommand{\cvmeasurementsentinelsection}{%
    \cvsection{Measurement Sentinel}%
    \cvmeasurementsentinelevent
}

% Measure section chrome as its observed marginal contribution immediately
% before the same event primitive used in production.
\newcommand{\cvsetmeasurementsectionchromebox}[1]{%
    \cvsetflowcontentbox{\cvmeasurementcandidatebox}{%
        \cvflowblockfitcontent{#1\cvmeasurementsentinelevent}%
    }%
    \cvsetflowcontentbox{\cvmeasurementreferencebox}{%
        \cvmeasurementsentinelevent
    }%
    \dimen0=\dimexpr
        \ht\cvmeasurementcandidatebox+\dp\cvmeasurementcandidatebox
        -\ht\cvmeasurementreferencebox-\dp\cvmeasurementreferencebox
        +1pt
        \relax
    \setbox\cvmeasurementbox=\vbox to \dimen0{\vfil}%
}

% Measure the marginal contribution of appending a bullet to the shared
% production itemize. The sentinel cancels top/bottom list chrome.
\newcommand{\cvsetmeasurementitembox}[1]{%
    \cvsetflowcontentbox{\cvmeasurementbox}{\cveventitem{#1}}%
    \dimen0=\dimexpr
        \ht\cvmeasurementbox+\dp\cvmeasurementbox+3pt
        \relax
    \setbox\cvmeasurementbox=\vbox to \dimen0{\vfil}%
}

% An event without permanent link bullets still acquires one-time itemize
% chrome as soon as its first selected experience item is rendered.
\newcommand{\cvsetmeasurementexperiencechromebox}[1]{%
    % The bottom half of the production list spacing is charged once here;
    % each marginal item carries its own leading/inter-item spacing.
    \cvsetmeasurementbox{#1}%
    \dimen0=\dimexpr
        \ht\cvmeasurementbox+\dp\cvmeasurementbox+3pt
        \relax
    \setbox\cvmeasurementbox=\vbox to \dimen0{\vfil}%
}

\NewEnviron{flowblock}{%
    \par

    % Measure both the raw body and the exact fit representation.
    \cvsetflowcontentbox{\flowbox}{\BODY}%
    \cvsetflowcontentbox{\flowfitbox}{\cvflowblockfitcontent{\BODY}}%

    \setlength{\flowheight}{%
        \dimexpr\ht\flowfitbox+\dp\flowfitbox\relax
    }

    % Remaining vertical space on current page
    \dimen0=\dimexpr\pagegoal-\pagetotal\relax

    % If the complete fit representation fits here, use the measured box.
    \ifdim\flowheight>\dimen0
        \newpage
        \cvflowblocknewpageskip

        % Large blocks become breakable.
        \dimen2=\dimexpr\ht\flowbox+\dp\flowbox\relax
        \ifdim\dimen2>\textheight
            \BODY
        \else
            \usebox{\flowbox}%
        \fi

        \par
        \hspace{1em plus 1fill}\allowbreak
    \else
        \usebox{\flowfitbox}%
    \fi
}
