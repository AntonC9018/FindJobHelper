\newcommand{\maybeitemize}[1]{%
  \def\temp{#1}%
  \ifx\temp\empty
    % do nothing, no items
  \else
    \begin{itemize}[topsep=3pt, partopsep=0pt, itemsep=3pt, parsep=0pt]
      #1
    \end{itemize}
  \fi
}

\newcommand{\maybeindent}[1]{%
  \def\temp{#1}%
  \ifx\temp\empty
  \else
    \par\hspace{1.5em}#1\par
  \fi
}

% creates a stretched box as cv entry headline followed by some paragraphs about
% the work you did
% param 1:	event time i.e. 2014 or 2011-2014 etc.
% param 2:	event name (what did you do?)
% param 3:	institution (where did you work / study)
% param 4:	list of paragraphs
% param 5:	just text before the bullet points
%
\newcommand{\cvevent}[5]
{
\begin{tabular*}{1\textwidth}{p{13.6cm}  x{3.9cm}}
    \textbf{#2}%
    \if\relax\detokenize{#3}\relax
    \else
        \ - \textcolor{bgcol}{#3}%
    \fi
    &
    \vspace{2.5pt}\textcolor{sectcol}{#1}
\end{tabular*}

\vspace{-8pt}
\textcolor{softcol}{\hrule}
\vspace{6pt}

\maybeindent{#5}

\maybeitemize{#4}

\vspace{6pt}
}

\usepackage{varwidth}
\usepackage{needspace}
\usepackage{environ}
\newsavebox{\flowbox}
\newlength{\flowheight}

\NewEnviron{flowblock}{%
    \par

    % Measure content
    \sbox{\flowbox}{%
        \begin{varwidth}{\linewidth}
            \BODY
        \end{varwidth}
    }%

    \setlength{\flowheight}{%
        \dimexpr\ht\flowbox+\dp\flowbox\relax
    }

    % Remaining vertical space on current page
    \dimen0=\dimexpr\pagegoal-\pagetotal\relax

    % If it fits here:
    \ifdim\flowheight<\dimen0
        \addvspace{\dimexpr\parskip + 1\baselineskip\relax}
    \else
        % If moving to next page, add larger top spacing
        \newpage
        \vspace*{1.5\baselineskip}
    \fi

    % Large blocks become breakable
    \ifdim\flowheight>\textheight
        \BODY
    \else
        \usebox{\flowbox}
    \fi

    \par

    \hspace{1em plus 1fill}\allowbreak
}
